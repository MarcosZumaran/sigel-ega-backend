<?php

namespace App\Services;

use App\DTOs\ApoderadoDTO;
use App\Exceptions\NotFoundException;
use App\Models\Apoderado;
use App\Repositories\ApoderadoRepository;
use Illuminate\Support\Collection;

class ApoderadoService
{
    public function __construct(private readonly ApoderadoRepository $repository)
    {
    }

    public function getAll(): Collection
    {
        return $this->repository->findAll();
    }

    public function getById(int $id): Apoderado
    {
        $model = $this->repository->findById($id);

        if (! $model) {
            throw new NotFoundException('Apoderado', $id);
        }

        return $model;
    }

    public function create(ApoderadoDTO $dto): Apoderado
    {
        return $this->repository->save(new Apoderado($dto->toArray()));
    }

    public function update(int $id, ApoderadoDTO $dto): Apoderado
    {
        $model = $this->getById($id);
        $model->fill($dto->toArray());

        return $this->repository->save($model);
    }

    public function delete(int $id): void
    {
        $model = $this->getById($id);

        if ($model->padres()->exists() || $model->estudiantes()->exists()) {
            throw new \App\Exceptions\EnUsoException('No se puede eliminar el Apoderado; tiene padres o estudiantes asociados.');
        }

        $this->repository->delete($model);
    }

    public function getPadres(int $id): Collection
    {
        return $this->getById($id)->padres;
    }

    public function getEstudiantes(int $id): Collection
    {
        return $this->getById($id)->estudiantes;
    }
    public function restore(int $id): Apoderado
    {
        $model = $this->repository->findTrashedById($id);
        if (! $model || ! $model->trashed()) {
            throw new NotFoundException('Apoderado eliminado', $id);
        }
        $this->repository->restore($model);
        return $model->refresh();
    }
}

    }

        return $this->repository->save(new AsistenciaPersonal($dto->toArray()));

        $model = $this->getById($id);

        $model = $this->repository->findTrashedById($id);
        if (! $model || ! $model->trashed()) {
            throw new NotFoundException('Asistencia de personal eliminada', $id);
        }
        $this->repository->restore($model);

        return $model->refresh();
    }

    public function marcarPropia(int $authUserId, AsistenciaPersonalDTO $dto): AsistenciaPersonal
    {
        // resolver personal del usuario autenticado
        $personal = \App\Models\Personal::where('user_id', $authUserId)->first();
        if (! $personal) {
            throw new NotFoundException('Personal vinculado al usuario', $authUserId);
        }
        $data = $dto->toArray();
        $data['personal_id'] = $personal->id;
        $data['registrado_por'] = $authUserId;

        return $this->repository->save(new AsistenciaPersonal($data));
    }

    public function justificar(int $id, ?string $descripcion, array $archivos = []): AsistenciaPersonal
    {
        $model = $this->getById($id);
        if ($descripcion !== null) {
            $model->descripcion_justificacion = $descripcion;
            $model->save();
        }
        foreach ($archivos as $file) {
            $path = $file->store('justificaciones_personal/' . $model->id, 'local');
            AsistenciaPersonalArchivo::create([
                'asistencia_id' => $model->id,
                'ruta_archivo' => $path,
                'nombre_original' => $file->getClientOriginalName(),
                'mime' => $file->getMimeType(),
                'tamano' => $file->getSize(),
            ]);
        }

        return $model->load(['archivos', 'personal', 'registradoPor']);
    }
}
<?php

namespace App\Services;

use App\DTOs\AsistenciaDTO;
use App\Exceptions\NotFoundException;
use App\Models\Asistencia;
use App\Models\Matricula;
use App\Repositories\AsistenciaRepository;
use Illuminate\Support\Collection;
use Illuminate\Validation\ValidationException;

class AsistenciaService
{
    public function __construct(private readonly AsistenciaRepository $repository)
    {
    }

    public function getAll(): Collection
    {
        return $this->repository->findAll();
    }

    public function getById(int $id): Asistencia
    {
        $model = $this->repository->findById($id);

        if (! $model) {
            throw new NotFoundException('Asistencia', $id);
        }

        return $model;
    }

    public function create(AsistenciaDTO $dto): Asistencia
    {
        $this->validar($dto->toArray());
        $this->assertPuedeGestionarAsistencia($dto->toArray());
        return $this->repository->save(new Asistencia($dto->toArray()));
    }

    public function update(int $id, AsistenciaDTO $dto): Asistencia
    {
        $model = $this->getById($id);
        $data = array_merge($model->toArray(), array_filter($dto->toArray(), fn ($v) => ! is_null($v)));
        $this->validar($data);
        $this->assertPuedeGestionarAsistencia($data, $model);
        $model->fill($dto->toArray());

        return $this->repository->save($model);
    }

    public function delete(int $id): void
    {
        $model = $this->getById($id);
        $this->assertPuedeGestionarAsistencia($model->toArray(), $model);
        $this->repository->delete($model);
    }

    private function assertPuedeGestionarAsistencia(array $data, ?Asistencia $existing = null): void
    {
        $user = auth()->user();
        if (! $user) return;
        if ((int) $user->rol_id === 1) return;
        $matId = $data['matricula_id'] ?? $existing?->matricula_id;
        if (! $matId) throw new \App\Exceptions\EnUsoException('Solo el administrador puede gestionar asistencias sin matrícula.');
        $mat = \App\Models\Matricula::with('seccion')->find($matId);
        if (! $mat || ! $mat->seccion) throw new \App\Exceptions\EnUsoException('Matrícula o sección no encontrada.');
        $docenteId = $mat->seccion->docente_id;
        $personal = \App\Models\Personal::where('user_id', $user->id)->first();
        $ok = false;
        if ($personal && $personal->categoria === 'docente' && (int) $personal->id === (int) $docenteId) $ok = true;
        if (! $ok) throw new \App\Exceptions\EnUsoException('Solo el docente de la sección o el administrador puede gestionar asistencias de esa matrícula.');
    }

    private function validar(array $data): void
    {
        $matId = $data['matricula_id'] ?? null;
        $fecha = $data['fecha'] ?? null;
        if ($matId && $fecha) {
            $mat = Matricula::with('periodo')->find($matId);
            if ($mat && $mat->periodo && $mat->periodo->fecha_inicio && $mat->periodo->fecha_fin) {
                if ($fecha < $mat->periodo->fecha_inicio || $fecha > $mat->periodo->fecha_fin) {
                    throw ValidationException::withMessages(['fecha' => 'La fecha de asistencia debe estar dentro del periodo de la matrícula ('.$mat->periodo->fecha_inicio.' a '.$mat->periodo->fecha_fin.').']);
                }
            }
            // Nota: se permite fecha futura dentro del periodo para simulación de apertura de año
            // if ($fecha > date('Y-m-d')) {
            //     throw ValidationException::withMessages(['fecha' => 'La fecha de asistencia no puede ser futura.']);
            // }
        }
    }

    public function restore(int $id): Asistencia
    {
        $model = $this->repository->findTrashedById($id);
        if (! $model || ! $model->trashed()) {
            throw new NotFoundException('Asistencia eliminado', $id);
        }
        $this->repository->restore($model);
        return $model->refresh();
    }
}

class CalificacionService
{
    public function __construct(private readonly CalificacionRepository $repository)
    {
    }

    public function getAll(): Collection
    {
        return $this->repository->findAll();
    }

    public function getById(int $id): Calificacion
    {
        $model = $this->repository->findById($id);

        if (! $model) {
            throw new NotFoundException('Calificación', $id);
        }

        return $model;
    }

    public function create(CalificacionDTO $dto): Calificacion
    {
        $this->validar($dto->toArray());
        $this->assertPuedeGestionarCalificacion($dto->toArray());
        return $this->repository->save(new Calificacion($dto->toArray()));
    }

    public function update(int $id, CalificacionDTO $dto): Calificacion
    {
        $model = $this->getById($id);
        $data = array_merge($model->toArray(), array_filter($dto->toArray(), fn ($v) => ! is_null($v)));
        $this->validar($data);
        $this->assertPuedeGestionarCalificacion($data, $model);
        $model->fill($dto->toArray());

        return $this->repository->save($model);
    }

    public function delete(int $id): void
    {
        $model = $this->getById($id);
        $this->assertPuedeGestionarCalificacion($model->toArray(), $model);
        $this->repository->delete($model);
    }

    private function assertPuedeGestionarCalificacion(array $data, ?\App\Models\Calificacion $existing = null): void
    {
        $user = auth()->user();
        if (! $user) return;
        if ((int) $user->rol_id === 1) return;
        // DOCENTE: solo su sección
        $matId = $data['matricula_id'] ?? $existing?->matricula_id;
        if (! $matId) throw new \App\Exceptions\EnUsoException('Solo el administrador puede gestionar calificaciones sin matrícula.');
        $mat = \App\Models\Matricula::with('seccion')->find($matId);
        if (! $mat || ! $mat->seccion) throw new \App\Exceptions\EnUsoException('Matrícula o sección no encontrada.');
        $docenteId = $mat->seccion->docente_id;
        // buscar personal/docente del usuario
        $personal = \App\Models\Personal::where('user_id', $user->id)->first();
        $docenteMatch = false;
        if ($personal && $personal->categoria === 'docente') {
            // personal.id 1..3 mapea a docente.id 1..3 en seed
            $docenteMatch = ((int) $personal->id === (int) $docenteId) || ((int) $personal->id <=3 && (int) $docenteId <=3 && (int) $personal->id === (int) $docenteId);
            // fallback: si personal es docente 3, solo sección docente_id 3
            if (! $docenteMatch) {
                // también buscar docente por dni si personal no mapea
                $doc = \App\Models\Docente::find($docenteId);
                if ($doc && $personal->dni === $doc->dni) $docenteMatch = true;
            }
        } else {
            $doc = \App\Models\Docente::where('dni', $user->email)->first(); // fallback email
        }
        if (! $docenteMatch) {
            throw new \App\Exceptions\EnUsoException('Solo el docente de la sección o el administrador puede gestionar calificaciones de esa matrícula.');
        }
    }

    private function validar(array $data): void
    {
        $esC = $data['es_nota_c'] ?? false;
        $motivo = $data['motivo_nota_c'] ?? null;
        $nivel = $data['nivel_logro'] ?? null;
        $nota = $data['nota'] ?? null;

        $esCBool = filter_var($esC, FILTER_VALIDATE_BOOLEAN, FILTER_NULL_ON_FAILURE) ?? (bool) $esC;

        // CNEB literal: si nivel_logro == C es equivalente a es_nota_c
        if ($nivel === 'C') $esCBool = true;
        if ($esCBool && empty(trim((string) $motivo))) {
            throw ValidationException::withMessages(['motivo_nota_c' => 'La nota C / nivel C exige motivo (¿por qué C? competencia no lograda).']);
        }
        if ($nivel && ! in_array($nivel, ['AD','A','B','C'], true)) {
            throw ValidationException::withMessages(['nivel_logro' => 'Nivel debe ser AD, A, B o C (CNEB).']);
        }
        // coherencia literal vs vigesimal: si escala literal, nota debe ser nula o ignorada
        if (($data['escala'] ?? null) === 'literal' && $nota !== null && $nivel === null) {
            // permitir pero preferir nivel_logro
        }
        if ($nivel === 'C' && $nota !== null) {
            // C no lleva nota numérica
            // no bloqueamos, solo validamos motivo ya hecho
        }
    }

    public function restore(int $id): Calificacion
    {
        $model = $this->repository->findTrashedById($id);
        if (! $model || ! $model->trashed()) {
            throw new NotFoundException('Calificacion eliminado', $id);
        }
        $this->repository->restore($model);
        return $model->refresh();
    }
}

<?php

namespace App\Services;

use App\DTOs\EstudianteDTO;
use App\Exceptions\NotFoundException;
use App\Models\Estudiante;
use App\Models\Grado;
use App\Repositories\EstudianteRepository;
use Illuminate\Support\Collection;
use Illuminate\Validation\ValidationException;

class EstudianteService
{
    public function __construct(private readonly EstudianteRepository $repository)
    {
    }

    public function getAll(): Collection
    {
        return $this->repository->findAll();
    }

    public function getById(int $id): Estudiante
    {
        $model = $this->repository->findById($id);

        if (! $model) {
            throw new NotFoundException('Estudiante', $id);
        }

        return $model;
    }

    public function create(EstudianteDTO $dto): Estudiante
    {
        $this->validar($dto->toArray());
        return $this->repository->save(new Estudiante($dto->toArray()));
    }

    public function update(int $id, EstudianteDTO $dto): Estudiante
    {
        $model = $this->getById($id);
        $data = array_merge($model->toArray(), array_filter($dto->toArray(), fn ($v) => ! is_null($v)));
        $this->validar($data);
        $model->fill($dto->toArray());

        return $this->repository->save($model);
    }

    private function validar(array $data): void
    {
        $fn = $data['fecha_nacimiento'] ?? null;
        if ($fn) {
            if ($fn > date('Y-m-d')) {
                throw ValidationException::withMessages(['fecha_nacimiento' => 'La fecha de nacimiento no puede ser futura.']);
            }
            $edad = (int) ((time() - strtotime($fn)) / 31557600);
            if ($edad < 3 || $edad > 25) {
                throw ValidationException::withMessages(['fecha_nacimiento' => 'Edad fuera de rango escolar (3-25 años). Verifique fecha.']);
            }
        }
        $nivelId = $data['nivel_id'] ?? null;
        $gradoId = $data['grado_id'] ?? null;
        if ($nivelId && $gradoId) {
            $grado = Grado::find($gradoId);
            if ($grado && (int) $grado->nivel_id !== (int) $nivelId) {

namespace App\Services;

use App\DTOs\MatriculaDTO;
use App\Exceptions\NotFoundException;
use App\Models\Estudiante;
use App\Models\Matricula;
use App\Models\Periodo;
use App\Models\Seccion;
use App\Repositories\MatriculaRepository;
use Illuminate\Support\Collection;
use Illuminate\Validation\ValidationException;

class MatriculaService
{
    public function __construct(private readonly MatriculaRepository $repository)
    {
    }

    public function getAll(): Collection
    {
        return $this->repository->findAll();
    }

    public function getById(int $id): Matricula
    {
        $model = $this->repository->findById($id);

        if (! $model) {
            throw new NotFoundException('Matrícula', $id);
        }

        return $model;
    }

    public function create(MatriculaDTO $dto): Matricula
    {
        $this->assertPuedeGestionarMatricula();
        $this->validarLogicaEscuela($dto->toArray());
        return $this->repository->save(new Matricula($dto->toArray()));
    }

    public function update(int $id, MatriculaDTO $dto): Matricula
    {
        $this->assertPuedeGestionarMatricula();
        $model = $this->getById($id);
        $data = array_merge($model->toArray(), array_filter($dto->toArray(), fn ($v) => ! is_null($v)));
        $this->validarLogicaEscuela($data, $id);
        $model->fill($dto->toArray());

        return $this->repository->save($model);
    }

    public function delete(int $id): void
    {
        $this->assertPuedeGestionarMatricula();
        $model = $this->getById($id);

        $bloqueos = ['calificaciones', 'asistencias'];

        foreach ($bloqueos as $relacion) {
            if ($model->{$relacion}()->exists()) {
                throw new \App\Exceptions\EnUsoException('No se puede eliminar el Matrícula; tiene registros asociados.');
            }
        }

        $this->repository->delete($model);
    }

    private function assertPuedeGestionarMatricula(): void
    {
        $user = auth()->user();
        if (! $user) return;
        if ((int) $user->rol_id === 1) return;
        // Solo ADMIN gestiona matrículas (secretaría/dirección)
        throw new \App\Exceptions\EnUsoException('Solo el administrador puede gestionar matrículas.');
    }

    private function validarLogicaEscuela(array $data, ?int $ignoreId = null): void
    {
        $estudianteId = $data['estudiante_id'] ?? null;
        $seccionId = $data['seccion_id'] ?? null;
        $periodoId = $data['periodo_id'] ?? null;
        $fecha = $data['fecha'] ?? null;

        if ($estudianteId && $seccionId) {
            $est = Estudiante::find($estudianteId);
            $sec = Seccion::with('grado')->find($seccionId);
            if ($est && $sec && $est->nivel_id && $sec->grado) {
                if ((int) $est->nivel_id !== (int) $sec->grado->nivel_id) {
                    throw ValidationException::withMessages(['seccion_id' => 'El nivel del estudiante no coincide con el nivel de la sección.']);
                }
            }
            if ($est && $sec && $est->grado_id && $sec->grado_id) {
                // opcional: grado del estudiante debe ser <= grado de la sección en promoción, no bloqueamos actualización si ya matriculado
            }
            // vacantes
            if ($sec && $sec->vacantes > 0) {
                $q = Matricula::where('seccion_id', $seccionId)->whereNull('deleted_at');
                if ($ignoreId) $q->where('id', '!=', $ignoreId);
                if ($q->count() >= $sec->vacantes) {
                    throw ValidationException::withMessages(['seccion_id' => 'La sección ya alcanzó su límite de vacantes ('.$sec->vacantes.').']);
                }
            }
        }

        if ($periodoId && $fecha) {
            $per = Periodo::find($periodoId);
            if ($per && $per->fecha_inicio && $per->fecha_fin) {
                if ($fecha < $per->fecha_inicio || $fecha > $per->fecha_fin) {
                    throw ValidationException::withMessages(['fecha' => 'La fecha de matrícula debe estar dentro del periodo ('.$per->fecha_inicio.' a '.$per->fecha_fin.').']);
                }
            }
        }
    }

    public function restore(int $id): Matricula
    {
        $model = $this->repository->findTrashedById($id);
        if (! $model || ! $model->trashed()) {
            throw new NotFoundException('Matricula eliminado', $id);
        }
        $this->repository->restore($model);
        return $model->refresh();
    }
}

{
    public function __construct(private readonly NecesidadEspecialRepository $repository) {}

    public function getAll(): Collection
    {
        return $this->repository->findAll();
    }

    public function getById(int $id): NecesidadEspecial
    {
        $model = $this->repository->findById($id);
        if (! $model) throw new NotFoundException('Necesidad especial', $id);
        return $model;
    }

    public function create(NecesidadEspecialDTO $dto): NecesidadEspecial
    {
        $data = $dto->toArray();
        $this->validar($data);
        // unique estudiante_id
        if (NecesidadEspecial::where('estudiante_id', $data['estudiante_id'])->exists()) {
            throw ValidationException::withMessages(['estudiante_id' => 'El estudiante ya tiene un registro de necesidad especial.']);
        }
        return $this->repository->save(new NecesidadEspecial($data));
    }

    public function update(int $id, NecesidadEspecialDTO $dto): NecesidadEspecial
    {
        $model = $this->getById($id);
        $data = array_merge($model->toArray(), array_filter($dto->toArray(), fn($v)=>!is_null($v)));
        $this->validar($data, $id);
        // si cambia estudiante_id validar duplicado
        if (isset($dto->toArray()['estudiante_id']) && $dto->toArray()['estudiante_id'] != $model->estudiante_id) {
            if (NecesidadEspecial::where('estudiante_id', $dto->toArray()['estudiante_id'])->where('id','!=',$id)->exists()) {
                throw ValidationException::withMessages(['estudiante_id' => 'El estudiante ya tiene un registro de necesidad especial.']);
            }
        }
        $model->fill($dto->toArray());
        return $this->repository->save($model);
    }

    public function delete(int $id): void
    {
        $model = $this->getById($id);
        $this->repository->delete($model);
    }

    public function restore(int $id): NecesidadEspecial
    {
        $model = $this->repository->findTrashedById($id);
        if (! $model || ! $model->trashed()) throw new NotFoundException('Necesidad especial eliminada', $id);
        $this->repository->restore($model);
        return $model->refresh();
    }

    private function validar(array $data, ?int $ignoreId=null): void
    {
        $servicio = $data['servicio'] ?? null;
        $estado = $data['estado_seho'] ?? null;
        $codigo = $data['codigo_seho'] ?? null;
        $nombre = $data['nombre_seho'] ?? null;
        $fechaSol = $data['fecha_solicitud'] ?? null;
        $fechaReinc = $data['fecha_reincorporacion'] ?? null;
        $reinc = $data['reincorporado'] ?? false;

        if ($servicio === 'SEHO') {
            if (!$estado) throw ValidationException::withMessages(['estado_seho' => 'SEHO exige estado_seho (pendiente/en_atencion/atendido/reincorporado).']);
            if (!$codigo) throw ValidationException::withMessages(['codigo_seho' => 'SEHO exige codigo_seho.']);
            if (!$nombre) throw ValidationException::withMessages(['nombre_seho' => 'SEHO exige nombre_seho (hospital/CREBE).']);
        }
        if ($fechaSol && $fechaReinc && $fechaReinc < $fechaSol) {
            throw ValidationException::withMessages(['fecha_reincorporacion' => 'La reincorporación no puede ser anterior a la solicitud.']);
        }
        if ($reinc && !$fechaReinc) {
            throw ValidationException::withMessages(['fecha_reincorporacion' => 'Reincorporado exige fecha_reincorporacion.']);
        }
        if ($fechaSol && $fechaSol > date('Y-m-d')) {
            throw ValidationException::withMessages(['fecha_solicitud' => 'Fecha de solicitud no puede ser futura.']);
        }
    }
}

namespace App\Services;

use App\DTOs\ReporteDTO;
use App\Exceptions\NotFoundException;
use App\Models\Reporte;
use App\Repositories\ReporteRepository;
use Illuminate\Contracts\Pagination\LengthAwarePaginator;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;

class ReporteService
{
    public function __construct(private readonly ReporteRepository $repository) {}

    public function paginate(Request $request): LengthAwarePaginator
    {
        return $this->repository->paginate($request);
    }

    public function getById(int $id): Reporte
    {
        $m = $this->repository->findById($id);
        if (! $m) throw new NotFoundException('Reporte', $id);
        return $m;
    }

    public function generar(ReporteDTO $dto, string $origen = 'api'): Reporte
    {
        $data = $dto->toArray();
        // expira en 1 año por defecto (listado anual)
        $data['expira_en'] = now()->addYear();
        $data['estado'] = $origen === 'cache' ? 'cacheado' : 'generado';
        $data['generado_por'] = $data['generado_por'] ?? auth()->id();
        // hash para evitar duplicados del mismo día/tipo
        $data['hash'] = hash('sha256', ($data['tipo'] ?? '') . '|' . ($data['periodo_id'] ?? '') . '|' . ($data['seccion_id'] ?? '') . '|' . now()->toDateString());
        // ruta flexible: storage/app/reportes/{periodo}/{seccion}/{tipo}-{fecha}.{formato}
        $periodo = $data['periodo_id'] ?? 'general';
        $seccion = $data['seccion_id'] ?? 'todas';
        $formato = $data['formato'] ?? 'pdf';
        $data['ruta_archivo'] = "reportes/{$periodo}/{$seccion}/{$data['tipo']}-" . now()->format('Y-m-d_His') . ".{$formato}";

        // stub: crear archivo vacío (en prod aquí va el generador PDF/Excel)
        Storage::disk('local')->put($data['ruta_archivo'], "Reporte {$data['tipo']} generado {$data['hash']}");

        $model = new Reporte($data);
        return $this->repository->save($model);
    }

    public function cacheNocturno(): int
    {
        $periodoActivo = \App\Models\Periodo::where('activo', true)->first();
        if (! $periodoActivo) return 0;
        $secciones = $this->repository->seccionesActivas();
        $count = 0;
        foreach (['auxiliar', 'asistencia'] as $tipo) {
            foreach ($secciones as $sec) {
                $dto = ReporteDTO::fromArray([
                    'tipo' => $tipo,
                    'periodo_id' => $periodoActivo->id,
                    'seccion_id' => $sec->id,
                    'formato' => 'pdf',
                    'parametros' => ['origen' => 'cache_nocturno'],
                ]);
                $this->generar($dto, 'cache');
                $count++;
            }
        }

        return $count;
    }

    public function delete(int $id): void
    {
        $m = $this->getById($id);
        $this->repository->delete($m);
    }

    public function restore(int $id): Reporte
    {
        $m = $this->repository->findTrashedById($id);
        if (! $m || ! $m->trashed()) throw new NotFoundException('Reporte eliminado', $id);
        $this->repository->restore($m);
        return $m->refresh();
    }
}

<?php

namespace App\Services;

use App\DTOs\UserDTO;
use App\Exceptions\NotFoundException;
use App\Models\User;
use App\Repositories\UserRepository;
use Illuminate\Support\Collection;

class UserService
{
    public function __construct(private readonly UserRepository $repository)
    {
    }

    public function getAll(): Collection
    {
        return $this->repository->findAll();
    }

    public function getById(int $id): User
    {
        $model = $this->repository->findById($id);

        if (! $model) {
            throw new NotFoundException('Usuario', $id);
        }

        return $model;
    }

    public function create(UserDTO $dto): User
    {
        $this->assertAdminParaRolAdmin($dto->toArray());
        return $this->repository->save(new User($dto->toArray()));
    }

    public function update(int $id, UserDTO $dto): User
    {
        $this->assertAdminParaRolAdmin($dto->toArray());
        $model = $this->getById($id);
        $model->fill($dto->toArray());

        return $this->repository->save($model);
    }

    private function assertAdminParaRolAdmin(array $data): void
    {
        if (! isset($data['rol_id']) || $data['rol_id'] === null) return;
        if ((int) $data['rol_id'] === 1) {
            $user = auth()->user();
            if (! $user || (int) $user->rol_id !== 1) {
                throw new \App\Exceptions\EnUsoException('Solo el administrador puede asignar rol ADMIN.');
            }
        }
    }

    public function delete(int $id): void
    {
        $model = $this->getById($id);

        $bloqueos = ['documentos'];

        foreach ($bloqueos as $relacion) {
            if ($model->{$relacion}()->exists()) {
                throw new \App\Exceptions\EnUsoException('No se puede eliminar el Usuario; tiene registros asociados.');
            }
        }

        $this->repository->delete($model);
    }
    public function restore(int $id): User
    {
        $model = $this->repository->findTrashedById($id);
        if (! $model || ! $model->trashed()) {
            throw new NotFoundException('User eliminado', $id);
        }
        $this->repository->restore($model);
        return $model->refresh();
    }
}
